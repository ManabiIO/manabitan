def equal_number(a, b):

def ast_blocks(raw):
    text = raw.decode()
    lines = text.splitlines(keepends=True)
    result = {}
    for node in ast.parse(text).body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            name = node.name
        elif isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
            name = node.targets[0].id
        else:
            continue
        result[name] = ''.join(lines[node.lineno - 1:node.end_lineno]).encode()
    return result

def expected_schedule(replication):
    result, counts = [], {'A': 0, 'B': 0}
    for index, arm in enumerate('ABBA' if replication == 1 else 'BAAB', 1):
        counts[arm] += 1
        result.append({'index': index, 'arm': arm, 'name': f'{arm}{counts[arm]}', 'pair': (index + 1) // 2})
    return result


def verify_sample(sample, previous_end):
    start, end = sample['startedAtMonotonicNs'], sample['finishedAtMonotonicNs']
    assert type(start) is int and type(end) is int and previous_end <= start <= end
    before, after = sample['discoveryBefore'], sample['discoveryAfter']
    stable = OBSERVER['tree_signature'](before) == OBSERVER['tree_signature'](after)
    assert sample['stableObservedProcessTree'] == stable
    reads, unavailable = sample['processes'], sample['unavailableProcesses']
    assert len({row['pid'] for row in reads + unavailable}) == len(reads) + len(unavailable)
    assert {row['pid'] for row in before['processes']} == {row['pid'] for row in reads + unavailable}
    for metric, total in [('pssKiB', 'readableProcessPssKiB'), ('privateKiB', 'readableProcessPrivateKiB')]:
        assert all(type(row[metric]) is int and row[metric] >= 0 for row in reads)
        assert sample[total] == sum(row[metric] for row in reads)
    after_by_pid = {row['pid']: row for row in after['processes']}
    exited = {row['pid'] for row in unavailable if OBSERVER['confirmed_exited_zombie'](row, after_by_pid.get(row['pid']))}
    assert {row['pid'] for row in sample['confirmedExitedZombies']} == exited
    unreadable = [row for row in unavailable if row['pid'] not in exited]
    assert unreadable == sample['unreadableAddressSpaces']
    roles = {row['role'] for row in reads}
    assert sorted(roles) == sample['observedRoles']
    failures = []
    if not before['browserSeedPids']: failures.append('no-browser-seed')
    if len(reads) < 2: failures.append('no-multiple-process-coverage')
    if 'browser' not in roles: failures.append('no-readable-browser-process')
    if 'renderer' not in roles: failures.append('no-readable-renderer-process')
    if not stable: failures.append('process-tree-changed-during-scan')
    if unreadable: failures.append('unreadable-discovered-process-smaps')
    if any(row['errno'] not in OBSERVER['GONE_ERRORS'] for row in before['inventoryIdentityReadErrors'] + after['inventoryIdentityReadErrors']):
        failures.append('incomplete-process-identity-inventory')
    assert sample['qualificationFailures'] == failures
    assert sample['qualifiedForWholeTreePss'] == (not failures)
    return end


def recompute_window(memory, markers):
    assert markers == memory['windowMarkers']
    assert len(markers) == 2 and [row['event'] for row in markers] == ['start', 'end']
    assert all(isinstance(row['monotonicNs'], str) and row['monotonicNs'].isdigit() for row in markers)
    start, end = [int(row['monotonicNs']) for row in markers]
    assert end > start
    samples = memory['samples']
    last = 0
    for sample in samples:
        last = verify_sample(sample, last)
    overlapping = [sample for sample in samples if sample['finishedAtMonotonicNs'] >= start and sample['startedAtMonotonicNs'] <= end]
    inside = [sample for sample in overlapping if sample['startedAtMonotonicNs'] >= start and sample['finishedAtMonotonicNs'] <= end]
    covered = [sample for sample in inside if sample['qualifiedForWholeTreePss']]
    before = [sample for sample in samples if sample['qualifiedForWholeTreePss'] and sample['finishedAtMonotonicNs'] <= start]
    after = [sample for sample in samples if sample['qualifiedForWholeTreePss'] and sample['startedAtMonotonicNs'] >= end]
    pre_gap = (start - before[-1]['finishedAtMonotonicNs']) / 1e6 if before else None
    post_gap = (after[0]['startedAtMonotonicNs'] - end) / 1e6 if after else None
    midpoints = [start] + [(sample['startedAtMonotonicNs'] + sample['finishedAtMonotonicNs']) // 2 for sample in covered] + [end]
    max_gap = max((right - left) / 1e6 for left, right in zip(midpoints, midpoints[1:]))
    failures = []
    if pre_gap is None or pre_gap > 250: failures.append('missing-covered-pre-import-bracket')
    if post_gap is None or post_gap > 250: failures.append('missing-covered-post-import-bracket')
    if len(covered) < 5: failures.append('fewer-than-five-covered-import-scans')
    if max_gap > 250: failures.append('covered-import-scan-gap-over-250ms')
    if memory['benchmarkExitCode'] != 0: failures.append('benchmark-command-failed')
    if not any(sample['qualifiedForWholeTreePss'] for sample in overlapping):
        failures.append('no-stable-fully-read-browser-renderer-tree')
    if (any(sample['stableObservedProcessTree'] and sample['unreadableAddressSpaces'] for sample in overlapping) or
            any(row['smapsReadError']['errno'] not in OBSERVER['GONE_ERRORS'] for sample in overlapping for row in sample['unreadableAddressSpaces'])):
        failures.append('process-smaps-coverage-unqualified')
    if any('incomplete-process-identity-inventory' in sample['qualificationFailures'] for sample in overlapping):
        failures.append('process-identity-coverage-unqualified')
    failures = list(dict.fromkeys(failures))
    qualified = not failures
    assert memory['qualificationFailures'] == failures and memory['qualifiedForWholeTreePss'] == qualified
    assert memory['authoritativeTiming'] is False
    assert memory['allObservedSampleCount'] == len(samples)
    assert memory['overlappingImportSampleCount'] == len(overlapping)
    assert memory['fullyContainedImportSampleCount'] == len(inside)
    assert memory['stableFullyReadImportSampleCount'] == len(covered)
    assert memory['sampleCount'] == len(overlapping)
    equal_number(memory['preImportBracketGapMs'], pre_gap)
    equal_number(memory['postImportBracketGapMs'], post_gap)
    equal_number(memory['maximumCoveredSampleGapMs'], max_gap)
    equal_number(memory['windowDurationMs'], (end - start) / 1e6)
    maxima = {}
    for metric, field in [('maxSampledPssKiB', 'readableProcessPssKiB'), ('maxSampledPrivateKiB', 'readableProcessPrivateKiB')]:
        maxima[metric] = max(sample[field] for sample in covered) if qualified else None
        assert memory[metric] == maxima[metric]
    return {'qualified': qualified, 'failures': failures, 'rawScans': len(samples),
            'coveredScans': len(covered), 'maximumCoveredGapMs': max_gap,
            'preImportBracketGapMs': pre_gap, 'postImportBracketGapMs': post_gap, **maxima}


def semantic_from_report(report, fixture):
    assert report['status'] == 'success' and report['skippedVerification'] is False
    assert report['failureReason'] == '' and report['skipReason'] == ''
    assert report['browserFlavor'] == 'chromium' and report['launchMode'] == 'headless'
    header = report['benchmark']
    for flag in ('authoritativeTiming', 'traceEnabled', 'phaseProfiling', 'phaseScreenshots', 'processSampling'):
        assert header[flag] is False
    assert header['productionImportDefaults'] is True and header['pinnedDictionaries'] is True
    assert header['importFlags'] is None and header['mediaCheck'] == 'all'
    validation = header['validation']
    assert validation['title'] == fixture['expectedTitle'] and validation['revision'] == fixture['revision']
    assert validation['termRows'] == fixture['termRows'] and validation['contentReadable'] is True
    probes = validation['probeTerms']
    assert validation['probeCount'] == len(probes) == len(set(probes)) == 12
    restart = validation['restart']
    assert restart['title'] == fixture['expectedTitle'] and restart['revision'] == fixture['revision']
    assert restart['termRows'] == fixture['termRows'] and restart['contentReadable'] is True
    assert restart['probeCount'] == 12 and restart['mediaFiles'] == fixture['mediaFiles']
    lookups = validation['lookup']['observations']
    assert len(lookups) == 72 and {(row['pass'], row['term']) for row in lookups} == {(i, term) for i in range(6) for term in probes}
    identities = {}
    for row in lookups:
        assert row['originalTextLength'] == len(row['term'].encode('utf-16-le')) // 2
        assert row['results'] > 0 and row['hasExactHeadwordMatch'] is True and len(row['sha256']) == 64
        identity = {key: row[key] for key in ('term', 'sha256', 'results', 'originalTextLength', 'hasExactHeadwordMatch')}
        if row['term'] in identities: assert identities[row['term']] == identity
        identities[row['term']] = identity
    media = validation['media']
    assert media['mode'] == 'all' and media['files'] == media['totalFiles'] == fixture['mediaFiles']
    assert media['bytes'] == media['totalBytes'] == fixture['mediaBytes']
    return {'title': validation['title'], 'revision': validation['revision'], 'termRows': validation['termRows'],
            'contentReadable': True, 'probeTerms': sorted(probes), 'publicResults': [identities[key] for key in sorted(identities)],
            'restart': restart, 'media': media}

    assert summary['fixture'] == LOCK[key[0]]

    assert read('plan.json') == plan and read('fixture.json') == summary['fixture']
    comparison = read('memory-comparison.json')
    assert comparison['authoritativeTiming'] is False and comparison['validationFailures'] == []
    raw_observations = read('observations.json')
    assert comparison['observations'] == raw_observations
    assert len(raw_observations) == len(summary['observations']) == len(summary['semanticProofs']) == 4

        assert semantic == read(f'{name}/semantic-receipt.json')
        assert sha(members[f'{name}/semantic-receipt.json']) == proof['semanticReceiptSha256']
        public_digest = sha(json.dumps(semantic['publicResults'], sort_keys=True, ensure_ascii=False).encode())
        assert public_digest == proof['publicResultSha256'] and proof['publicLookupObservationCount'] == 72
        assert proof['distinctProbeCount'] == 12 and proof['termRows'] == summary['fixture']['termRows']
        assert proof['contentReadable'] is True and proof['restartContentReadable'] is True
        assert proof['media'] == semantic['media']
        checked_observations.append({**scheduled, **checked})
        semantics.append(semantic)
    assert all(value == semantics[0] for value in semantics)
    qualified = all(row['qualified'] for row in checked_observations)
    assert comparison['everyObservationQualified'] == qualified
    recalculated_pairs = []
    for pair_index in (1, 2):

        both_qualified = selected['A']['qualified'] and selected['B']['qualified']

def main():
    summaries = json.loads((EVIDENCE / 'retrieved-artifacts.json').read_text())
    fulls = json.loads((EVIDENCE / 'full-evidence-index.json').read_text())
    cells, errors = [], []
    for retrieval in summaries:
        name = retrieval['artifact']['name'].replace('import-memory-summary-ef009-', 'import-memory-ef009-')
        matching = [full for full in fulls if full['artifact']['name'] == name]
        if not matching:
            continue
        assert len(matching) == 1
        try:
            cells.append(audit_cell(retrieval, matching[0]))
        except Exception as error:
            errors.append({'artifact': name, 'error': repr(error), 'traceback': traceback.format_exc()})
    observed = {(cell['dictionary'], cell['replication']) for cell in cells}
    assert len(observed) == len(cells)
    boots = [cell['hostIdentity']['bootId'] for cell in cells]
    hosts = [cell['hostIdentity']['hostname'] for cell in cells]
    runners = [cell['hostIdentity']['runnerName'] for cell in cells]
    if len(set(boots)) != len(boots): errors.append({'error': 'Boot ID reused across cells'})
    if len(set(runners)) != len(runners): errors.append({'error': 'Runner name reused across cells'})
    for dictionary in ('jmdict', 'jitendex', 'vndb'):
        selected = [cell for cell in cells if cell['dictionary'] == dictionary]
        for field in ('semanticReceiptSha256', 'publicResultSha256', 'fullMedia'):
            if len({json.dumps(cell[field], sort_keys=True) for cell in selected}) > 1:
                errors.append({'dictionary': dictionary, 'error': f'{field} differs across hosts'})
    for arm in 'AB':
        if len({cell['packageMemberMapSha256'][arm] for cell in cells}) > 1:
            errors.append({'arm': arm, 'error': 'Package member maps differ across hosts'})
    primary_path = EVIDENCE / 'independent-reaudit.json'
    primary = json.loads(primary_path.read_text()) if primary_path.exists() else None
    if primary:
        assert primary['workflowHead'] == WORKFLOW and str(primary['workflowRun']) == RUN_ID
        assert primary['protocolSha256'] == sha(PROTOCOL_BYTES)
        for cell in cells:
            matching = [row for row in primary['cells'] if (row['dictionary'], row['replication']) == (cell['dictionary'], cell['replication'])]
            assert len(matching) == 1
            peer = matching[0]
            assert peer['errors'] == [] and peer['evidenceVerified'] is True
            assert peer['metrics'] == cell['metrics'] and peer['pairs'] == cell['pairs']
            assert peer['hostIdentity'] == cell['hostIdentity']
            assert peer['qualified'] == cell['passesFrozenMemoryGate']
            assert peer['rawAudit']['rawScanCount'] == cell['rawScans']
            assert peer['rawAudit']['everyObservationQualified'] == cell['everyObservationQualified']
    complete = observed == EXPECTED and not errors
    result = {'schemaVersion': 1, 'authoritativeTiming': False,
              'workflowCommit': WORKFLOW, 'workflowTree': WORKFLOW_TREE, 'workflowRunId': RUN_ID, 'workflowAttempt': 1,
              'auditRunnerSha256': sha(Path(__file__).read_bytes()), 'protocolSha256': sha(PROTOCOL_BYTES),
              'retrievalMetadataSha256': {name: sha((EVIDENCE / name).read_bytes()) for name in ('retrieved-artifacts.json', 'full-evidence-index.json')},
              'observerSha256': {name: sha(value) for name, value in SOURCES.items()},
              'completeIndependentAudit': complete, 'passesFrozenSixCellMemoryGate': complete and all(cell['passesFrozenMemoryGate'] for cell in cells),
              'receivedSummaries': len(summaries), 'receivedFullBundles': len(fulls), 'auditedCells': len(cells),
              'observationsVerified': len(cells) * 4, 'publicLookupObservationsVerified': len(cells) * 288,
              'coverageQualifiedObservations': sum(row['qualified'] for cell in cells for row in cell['observations']),
              'memoryGatePassedCells': sum(cell['passesFrozenMemoryGate'] for cell in cells),
              'memoryGateFailedCells': [[cell['dictionary'], cell['replication']] for cell in cells if not cell['passesFrozenMemoryGate']],
              'rawScansRecomputed': sum(cell['rawScans'] for cell in cells),
              'distinctBootIds': len(set(boots)), 'distinctHostnames': len(set(hosts)), 'distinctRunnerNames': len(set(runners)),
              'hostnameReuseObserved': len(set(hosts)) != len(hosts),
              'primaryUnchangedAuditorReexecutionReceiptSha256': sha(primary_path.read_bytes()) if primary else None,
              'packageMemberMapsIdenticalAcrossCells': {arm: len(cells) == 6 and len({cell['packageMemberMapSha256'][arm] for cell in cells}) == 1 for arm in 'AB'},
              'errors': errors, 'missingCells': [list(key) for key in sorted(EXPECTED - observed)],
              'cells': sorted(cells, key=lambda cell: (cell['dictionary'], cell['replication'])),
              'limitations': ['Sampled non-atomic process-tree PSS/private maxima within the marked import window, not exact peaks.',
                             'Startup, media readback and restart are excluded from maxima; their semantics are checked separately.',
                             'Instrumented timing is explicitly non-authoritative; these results do not establish import speed.',
                             'Preserved package payload hashes/member maps identify the releases; large ZIP bytes remain in the original CI artifacts.',
                             'Distinct boot IDs and runner assignments establish distinct hosted instances, not physical hardware independence. Hostnames may be reused across those instances.']}
    OUTPUT.write_text(json.dumps(result, indent=2) + '\n')
    print(json.dumps(result, indent=2))


if __name__ == '__main__':
    main()
