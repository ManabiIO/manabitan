async function consume(chunk) {
    const plan = chunk.contentDedupPlan;
    assert(plan && chunk.contentUniqueIndexList && chunk.contentMetaList);
    const offsets = new Float64Array(plan.uniqueCount);
    const lengths = new Uint32Array(plan.uniqueCount);
    for (let i = 0; i < plan.uniqueCount; ++i) {
        const meta = (chunk.contentMetaByUniqueIndex ? i : plan.uniqueRowIndexes[i]) * 4;
        offsets[i] = contentBase + chunk.contentMetaList[meta];
        lengths[i] = chunk.contentMetaList[meta + 1];
    }
    const module = arms.A.module;
    const store = new module.TermRecordOpfsStore();
    const start = blocks.length;
    store._appendEncodedChunk = async (_state, firstId, count, _dict, base, payload, fields, format) => {
        // Retain private bytes before native buffers may be reused. This is the
        // only intercepted operation; grouping, parser, run splitting and field
        // encoding are the actual current implementations.
        blocks.push({firstId, count, base, payload: Uint8Array.from(payload), fields: Uint8Array.from(fields), format});
    };
    const recordChunk = module.createArtifactTermRecordChunkWithResolvedContent(
        {...chunk, dictionary: fixture.expectedTitle}, chunk.contentUniqueIndexList, offsets, lengths, chunk.preparedLookupIndexes);
    await store._encodeAndAppendArtifactChunkForState({}, recordChunk, rowCount + 1,
        new Float64Array(0), new Uint32Array(0), chunk.termRecordPreinternedPlan, 'raw', chunk.preparedLookupIndexes);
    groups.push({rows: chunk.rowCount, uniqueContent: plan.uniqueCount, runRows: blocks.slice(start).map(({count}) => count)});
    rowCount += chunk.rowCount;
    contentBase += chunk.contentBytesLength ?? chunk.contentBytesBuffer.byteLength;
}

// All traversal timings use the captured, current-encoder output for all rows.
// Enclosing-sidecar timing includes all checksums, allocation and copies. There
// is no instrumentation in the timed methods or production files.
let sink = 0;
function runValidator(arm, repetitions) {
    for (let repeat = 0; repeat < repetitions; ++repeat) {
        for (const block of blocks) {
            const value = arm.module.parseCompactRecordFields(block.fields, block.count);
            sink = (sink + value.scoreKeys[block.count - 1] + value.scores.length) | 0;
        }
    }
}
function runSidecars(arm, repetitions) {
    for (let repeat = 0; repeat < repetitions; ++repeat) {
        for (const block of blocks) {
            const value = arm.store._createLookupIndexChunk(block.firstId, block.count, block.base, block.payload, block.fields, block.format);
            sink = (sink + value.length + value[value.length - 1]) | 0;
        }
    }
}
