async function sourceIdentity(root) {
    const [commit, tree] = execFileSync('git', ['rev-parse', 'HEAD', 'HEAD^{tree}'], {cwd: root, encoding: 'utf8'}).trim().split('\n')
    const hashes = {}
    for (const file of sourceFiles) { hashes[file] = sha(await readFile(`${root}/${file}`)) }
    return {root, commit, tree, nativeSources: getDictionaryWasmSourceIdentity(root), hashes}
}

const options = {singleChunk: true, emitTermByteLists: false, computeContentHashes: true,

async function parseGroup(arm, sources) {
    const {parser, prep} = arms[arm]
    const hash = createHash('sha256')
    let rows = 0
    let indexBytes = 0
    let mediaRows = 0
    await parser.parseTermBankWithWasmColumnChunks(sources, 3, (chunk) => {
        const indexes = chunk.preparedLookupIndexes ?? prep.prepareTermLookupIndexesFromPreinternedPlan(chunk, null, {})?.indexes
        assert(indexes && prep.hasCompletePreparedTermLookupIndexes(indexes, chunk.rowCount), 'Incomplete lookup indexes')
        rows += chunk.rowCount
        for (const [key, entry] of indexes) {
            hash.update(key)
            hash.update(entry.bytes)
            indexBytes += entry.bytes.byteLength
            for (const view of [entry.preinternedPlan.stringLengths, entry.preinternedPlan.stringOffsets,
                entry.preinternedPlan.stringHashes, entry.preinternedPlan.stringsBuffer,
                entry.preinternedPlan.expressionIndexes, entry.preinternedPlan.readingIndexes]) {
                if (view) { updateView(hash, view) }
            }
        }
        updateView(hash, chunk.readingEqualsExpressionList)
        updateView(hash, chunk.scoreList)
        updateView(hash, chunk.sequenceList)
        const {contentMetaList, contentBytesBuffer, contentBytesBaseOffset} = chunk
        assert(contentMetaList && contentBytesBuffer && typeof contentBytesBaseOffset === 'number', 'Missing content slab')
        for (let i = 0; i < chunk.rowCount; ++i) {
            const offset = contentBytesBaseOffset + contentMetaList[i * 4]
            hash.update(contentBytesBuffer.subarray(offset, offset + contentMetaList[i * 4 + 1]))
            updateView(hash, contentMetaList.subarray(i * 4 + 1, i * 4 + 4))
        }
        for (const {index, row} of chunk.mediaRows) {
            ++mediaRows
            hash.update(JSON.stringify({index, expression: row.expression, reading: row.reading,
                definitionTags: row.definitionTags, rules: row.rules, score: row.score,
                sequence: row.sequence, termTags: row.termTags, glossaryJson: row.glossaryJson,
                glossaryMayContainMedia: row.glossaryMayContainMedia,
                termEntryContentHash1: row.termEntryContentHash1, termEntryContentHash2: row.termEntryContentHash2}))
            for (const view of [row.glossaryJsonBytes, row.termEntryContentBytes, row.expressionBytes, row.readingBytes]) {
                if (view) { updateView(hash, view) }
            }
        }
    }, 2048, options)
    const profile = parser.consumeLastTermBankWasmParseProfile()
    assert(profile, 'Missing parser profile')
    return {rows, indexBytes, mediaRows, digest: hash.digest('hex'),
        fusedParseAttempts: profile.fusedParseAttempts ?? 0,
        fusedParseFallbacks: profile.fusedParseFallbacks ?? 0,
        maxWasmHeapBytes: profile.maxWasmHeapBytes ?? null}
}
