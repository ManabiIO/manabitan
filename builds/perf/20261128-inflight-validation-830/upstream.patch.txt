+/**
+ * Validates current reservation columns without repeating invariant checks for
+ * every entry. Origin marks never cache validation; shared/accessor metadata
+ * and nonascending reservations use the complete generic validator.
+ * @param {number[]|Float64Array} offsets
+ * @param {number[]|Uint32Array} lengths
+ * @param {string|string[]} dictNames
+ * @param {InFlightTermContentSourceBatch['contentSpans']} spans
+ * @param {number} count
+ * @returns {boolean}
+ */
+function hasSimpleInFlightTermContentSources(offsets, lengths, dictNames, spans, count) {
+    if (
+        spans === null || !INTERNAL_PENDING_CONTENT_SPANS.has(spans) || count === 0 ||
+        typeof dictNames !== 'string' || typeof ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR?.get !== 'function'
+    ) { return false; }
+    const sourceBytes = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(spans, 'buffer')?.value);
+    const sourceOffsets = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(spans, 'offsets')?.value);
+    const sourceLengths = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(spans, 'lengths')?.value);
+    if (
+        TYPED_ARRAY_TAG_DESCRIPTOR?.get?.call(sourceBytes) !== 'Uint8Array' ||
+        Object.getPrototypeOf(sourceBytes) !== Uint8Array.prototype || !(sourceBytes instanceof Uint8Array) ||
+        Object.hasOwn(sourceBytes, 'byteLength') ||
+        TYPED_ARRAY_TAG_DESCRIPTOR?.get?.call(offsets) !== 'Float64Array' ||
+        Object.getPrototypeOf(offsets) !== Float64Array.prototype || !(offsets instanceof Float64Array) ||
+        TYPED_ARRAY_TAG_DESCRIPTOR?.get?.call(lengths) !== 'Uint32Array' ||
+        Object.getPrototypeOf(lengths) !== Uint32Array.prototype || !(lengths instanceof Uint32Array) ||
+        TYPED_ARRAY_TAG_DESCRIPTOR?.get?.call(sourceOffsets) !== 'Uint32Array' ||
+        Object.getPrototypeOf(sourceOffsets) !== Uint32Array.prototype || !(sourceOffsets instanceof Uint32Array) ||
+        sourceLengths !== lengths
+    ) { return false; }
+    for (const column of [offsets, lengths, sourceOffsets]) {
+        if (
+            !ArrayBuffer.isView(column) || Object.hasOwn(column, 'length') || Object.hasOwn(column, 'buffer')
+        ) { return false; }
+        const buffer = column.buffer;
+        try {
+            // Native getters prove private, fixed backing storage even when
+            // its prototype or visible capability properties were replaced.
+            ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR.get.call(buffer);
+            if (ARRAY_BUFFER_RESIZABLE_DESCRIPTOR?.get?.call(buffer) === true) { return false; }
+        } catch (_) {
+            return false;
+        }
+    }
+    if (sourceOffsets.length < offsets.length) { return false; }
+    let previousOffset = -1;
+    for (let i = 0; i < offsets.length; ++i) {
+        const offset = offsets[i];
+        // Uint32 source columns already prove nonnegative integral values, and
+        // the persisted lengths are the identical, synchronously stable column.
+        if (
+            !Number.isSafeInteger(offset) || offset <= previousOffset ||
+            lengths[i] > sourceBytes.byteLength - sourceOffsets[i]
+        ) { return false; }
+        previousOffset = offset;
+    }
+    return true;
+}
