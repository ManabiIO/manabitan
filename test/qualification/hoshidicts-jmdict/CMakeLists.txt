cmake_minimum_required(VERSION 3.22)
project(hoshidicts_jmdict_qualification LANGUAGES CXX)
set(HOSHIDICTS_SOURCE_DIR "" CACHE PATH "Pinned Hoshidicts source checkout")
if(NOT HOSHIDICTS_SOURCE_DIR)
  message(FATAL_ERROR "Set HOSHIDICTS_SOURCE_DIR to the pinned Hoshidicts checkout")
endif()
add_subdirectory("${HOSHIDICTS_SOURCE_DIR}" "hoshidicts-build")
find_package(OpenSSL REQUIRED)
add_executable(hoshidicts-import-lookup-bench hoshidicts-import-lookup-bench.cpp)
target_link_libraries(hoshidicts-import-lookup-bench PRIVATE hoshidicts OpenSSL::Crypto)
