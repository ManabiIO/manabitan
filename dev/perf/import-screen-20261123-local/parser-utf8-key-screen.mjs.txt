async function loadKeys(dictionary) {
    const spec = locks.dictionaries[dictionary]
    const archive = await readFile(path.join(cache, spec.cacheFile))
    assert.equal(sha256(archive), spec.sha256)
    const reader = new ZipReader(new Uint8ArrayReader(new Uint8Array(archive)), {useWebWorkers: false})
    const entries = (await reader.getEntries()).filter((e) => /^term_bank_\d+\.json$/.test(e.filename)).sort((a, b) => Number(a.filename.slice(10, -5)) - Number(b.filename.slice(10, -5)))
    const rows = []
    for (const entry of entries) {
        const at = entry.offset
        const start = at + 30 + archive.readUInt16LE(at + 26) + archive.readUInt16LE(at + 28)
        const bytes = archive.subarray(start, start + entry.compressedSize)
        const decoded = entry.compressionMethod === 0 ? bytes : inflateRawSync(bytes)
        assert.equal(decoded.length, entry.uncompressedSize)
        assert.equal(crc32(decoded), entry.signature)
        for (const row of JSON.parse(decoded.toString())) { rows.push([row[0], row[1], '', '', 0, [], 0, '']) }
    }
    assert.equal(rows.length, spec.termRows)
    await reader.close()
    return rows
}

async function setup(module, rows, bytes) {
    const wasm = (await WebAssembly.instantiate(module)).exports
    wasm.wasm_reset_heap()
    const input = wasm.wasm_alloc(bytes.length)
    const metas = wasm.wasm_alloc(rows * 68)
    const stringsCapacity = bytes.length
    const strings = wasm.wasm_alloc(stringsCapacity)
    const lengths = wasm.wasm_alloc(rows * 4)
    const offsets = wasm.wasm_alloc(rows * 8)
    const hashes = wasm.wasm_alloc(rows * 8)
    const expressions = wasm.wasm_alloc(rows * 4)
    const readings = wasm.wasm_alloc(rows * 4)
    let slots = 1
    while (slots < rows * 4) { slots *= 2 }
    const table = wasm.wasm_alloc(slots * 4)
    const count = wasm.wasm_alloc(4)
    new Uint8Array(wasm.memory.buffer, input, bytes.length).set(bytes)
    assert.equal(wasm.parse_term_bank_with_media_hints(input, bytes.length, metas, rows), rows)
    const hashSlots = new Uint32Array(wasm.memory.buffer, table, slots)
    const counts = new Uint32Array(wasm.memory.buffer, count, 1)
    let used = 0
    function run() {
        hashSlots.fill(0)
        counts[0] = 0
        used = wasm.build_term_string_plan(input, metas, 0, rows, strings, stringsCapacity, lengths, offsets, hashes, expressions, readings, table, slots, count)
        assert.ok(used >= 0)
    }
    function digest() {
        const hash = createHash('sha256')
        for (const [at, size] of [[strings, used], [lengths, counts[0] * 2], [offsets, counts[0] * 4], [hashes, counts[0] * 4], [expressions, rows * 4], [readings, rows * 4]]) {
            hash.update(new Uint8Array(wasm.memory.buffer, at, size))
        }
        return hash.digest('hex')
    }
    run()
    return {run, digest, stats: () => ({uniqueKeys: counts[0], keyBytes: used, wasmHeapBytes: wasm.memory.buffer.byteLength})}
}

            const order = block % 2 === 0 ? [0, 1, 0] : [1, 0, 1]
            const times = []
            for (const arm of order) {
                const start = performance.now()
                for (let iteration = 0; iteration < report.iterationsPerObservation; ++iteration) { arms[arm].run() }
                times.push(performance.now() - start)

            const outside = (times[0] + times[2]) / 2
            const ratio = block % 2 === 0 ? times[1] / outside : outside / times[1]

        }
