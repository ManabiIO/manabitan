import assert from 'node:assert/strict'
import {afterEach, describe, expect, test, vi} from 'vitest'
import {createArtifactTermRecordChunkWithResolvedContent, TermRecordOpfsStore} from '../ext/js/dictionary/term-record-opfs-store.js'

/** @typedef {ReturnType<typeof fixture>} Fixture */
/** @typedef {Parameters<TermRecordOpfsStore['_appendEncodedChunk']>} AppendArguments */

/**
 * A 256-row input splits at the offset boundary. Larger inputs split at the
 * existing 30K lookup limit, using the same compact fields as real imports.
 * @param {number} [count=256]
 * @returns {ReturnType<typeof createArtifactTermRecordChunkWithResolvedContent<{
 * dictionary: string, rowCount: number, dictionaryTotalRows: number,
 * expressionBytesList: Uint8Array[], readingBytesList: Uint8Array[],
 * readingEqualsExpressionList: Uint8Array, scoreList: Int32Array,
 * sequenceList: Int32Array,
 * termRecordPreinternedPlan: import('../ext/js/dictionary/term-record-preinterned-plan.js').PreinternedTermRecordPlan
 * }>>}
 */
function fixture(count = 256) {
    const stringsBuffer = new TextEncoder().encode('語')
    const uniqueCount = 256
    return createArtifactTermRecordChunkWithResolvedContent(
        {
            dictionary: 'Sliced resolved columns',
            rowCount: count,
            dictionaryTotalRows: 100_000,
            expressionBytesList: [],
            readingBytesList: [],
            readingEqualsExpressionList: new Uint8Array(count).fill(1),
            scoreList: Int32Array.from({length: count}, (_, i) => i % 7 - 3),
            sequenceList: new Int32Array(count).fill(-1),
            termRecordPreinternedPlan: {
                stringLengths: Uint16Array.of(stringsBuffer.byteLength),
                stringOffsets: Uint32Array.of(0),
                stringsBuffer,
                expressionIndexes: new Uint32Array(count),
                readingIndexes: new Uint32Array(count),
            },
        },
        Uint32Array.from({length: count}, (_, i) => i % uniqueCount),
        Float64Array.from({length: uniqueCount}, (_, i) => 2 ** 40 + i * 19 + (count === 256 && i >= 128 ? 0xffffffff : 0)),
        Uint32Array.from({length: uniqueCount}, (_, i) => i % 31),
    )
}

/**
 * Execute the real validation and encoders while intercepting only persistence.
 * Measure staging copies inside the field encoder, excluding lookup preparation.
 * @param {((args: AppendArguments, index: number) => void|Promise<void>)|null} [afterAppend=null]
 * @returns {{store: TermRecordOpfsStore, writes: AppendArguments[], copies: number[]}}
 */
function createStore(afterAppend = null) {
    const store = new TermRecordOpfsStore()
    /** @type {AppendArguments[]} */
    const writes = []
    /** @type {number[]} */
    const copies = []
    const encodeFields = store._encodeArtifactRecordFields.bind(store)
    vi.spyOn(store, '_encodeArtifactRecordFields').mockImplementation((...args) => {
        const copiedOffsets = vi.spyOn(Uint32Array.prototype, 'set')
        try {
            const result = encodeFields(...args)
            copies.push(copiedOffsets.mock.calls.length)
            return result
        } finally {
            copiedOffsets.mockRestore()
        }
    })
    vi.spyOn(store, '_appendEncodedChunk').mockImplementation(async (...args) => {
        writes.push(args)
        await afterAppend?.(args, writes.length - 1)
    })
    return {store, writes, copies}
}

/**
 * @param {TermRecordOpfsStore} store
 * @param {Fixture} chunk
 * @returns {ReturnType<TermRecordOpfsStore['_encodeAndAppendArtifactChunkForState']>}
 */
function encode(store, chunk) {
    return store._encodeAndAppendArtifactChunkForState(
        /** @type {Parameters<TermRecordOpfsStore['_encodeAndAppendArtifactChunkForState']>[0]} */ ({}),
        chunk,
        1,
        new Float64Array(0),
        new Uint32Array(0),
        chunk.termRecordPreinternedPlan,
        'raw',
    )
}

afterEach(() => { vi.restoreAllMocks() })

describe('resolved column origin through record splits', () => {
    test.each([256, 60_128])('keeps exact lookup/field bytes without staged field copies across %i rows', async (count) => {
        const chunk = fixture(count)
        const generic = createStore()
        await encode(generic.store, {...chunk})
        const candidate = createStore()
        await encode(candidate.store, chunk)
        assert.deepStrictEqual(candidate.writes, generic.writes)
        expect(candidate.writes.map((args) => args[2])).toStrictEqual(count === 256 ? [128, 128] : [30_000, 30_000, 128])
        expect(generic.copies).toStrictEqual(count === 256 ? [1, 1] : [1, 1, 1])
        expect(candidate.copies).toStrictEqual(count === 256 ? [0, 0] : [0, 0, 0])
        for (const args of candidate.writes) {
            const fields = /** @type {Uint8Array} */ (args[6])
            expect(fields.buffer.byteLength).toBe(fields.byteLength)
        }
    })

    test.each(['unproven', 'reference-replaced', 'shared', 'score-accessor', 'score-subclass'])('keeps %s source snapshots on the generic path', async (kind) => {
        let chunk = fixture()
        switch (kind) {
            case 'unproven': chunk = {...chunk}; break
            case 'reference-replaced': chunk.resolvedContentReferences = {...chunk.resolvedContentReferences}; break
            case 'shared': {
                const lengths = new Uint32Array(new SharedArrayBuffer(256 * 4))
                lengths.set(chunk.resolvedContentReferences.lengths)
                chunk.resolvedContentReferences.lengths = lengths
                break
            }
            case 'score-accessor': {
                const scores = chunk.scoreList
                Object.defineProperty(chunk, 'scoreList', {get: () => scores})
                break
            }
            default: {
                class Scores extends Int32Array {}
                chunk.scoreList = new Scores(chunk.scoreList)
                break
            }
        }
        const generic = createStore()
        await encode(generic.store, {...chunk})
        const candidate = createStore()
        await encode(candidate.store, chunk)
        assert.deepStrictEqual(candidate.writes, generic.writes)
        expect(candidate.copies).toStrictEqual([1, 1])
    })

    test('rechecks provenance when an awaited append replaces the producer reference object', async () => {
        const chunk = fixture()
        const candidate = createStore(async (_args, index) => {
            await Promise.resolve()
            if (index === 0) { chunk.resolvedContentReferences = {...chunk.resolvedContentReferences} }
        })
        const generic = createStore()
        await encode(generic.store, {...chunk})
        await encode(candidate.store, chunk)
        assert.deepStrictEqual(candidate.writes, generic.writes)
        expect(candidate.copies).toStrictEqual([0, 1])
    })

    test.each(['offset', 'length', 'index', 'accessor'])('rejects invalid %s changes after an awaited compact-field append', async (kind) => {
        const chunk = fixture()
        const columns = chunk.resolvedContentReferences
        const candidate = createStore(async (_args, index) => {
            await Promise.resolve()
            if (index !== 0) { return }
            if (kind === 'offset') { columns.offsets[128] = Number.NaN }
            if (kind === 'length') { columns.lengths[128] = 0xffffffff }
            if (kind === 'index') { columns.uniqueIndexList[128] = 256 }
            if (kind === 'accessor') {
                const offsets = new Float64Array(columns.offsets)
                offsets[128] = Number.NaN
                Object.defineProperty(columns, 'offsets', {get: () => offsets})
            }
        })
        await expect(encode(candidate.store, chunk)).rejects.toThrow(/Invalid (term content|resolved term content reference)/)
        expect(candidate.writes).toHaveLength(1)
        expect(candidate.copies).toStrictEqual([0])
    })

    test('rereads future runs after an await while keeping completed encoded fields owned', async () => {
        /** @type {Uint8Array|null} */
        let firstBytes = null
        const chunk = fixture()
        const candidate = createStore(async (args, index) => {
            await Promise.resolve()
            if (index !== 0) { return }
            firstBytes = new Uint8Array(/** @type {Uint8Array} */ (args[6]))
            chunk.scoreList.fill(-37, 128)
            chunk.resolvedContentReferences.lengths.fill(29)
        })
        await encode(candidate.store, chunk)
        expect(candidate.writes[0][6]).toEqual(firstBytes)
        const secondFields = /** @type {Uint8Array} */ (candidate.writes[1][6])
        const view = new DataView(secondFields.buffer, secondFields.byteOffset, secondFields.byteLength)
        expect(view.getInt32(16, true)).toBe(-37)
        expect(view.getUint16(20 + 128 * 4, true)).toBe(29)
        expect(candidate.copies).toStrictEqual([0, 0])
    })

    test('rechecks sliced score descriptors before the final synchronous encoder', async () => {
        const chunk = fixture()
        const candidate = createStore()
        const encodeChunk = candidate.store._encodeArtifactChunkRecords.bind(candidate.store)
        vi.spyOn(candidate.store, '_encodeArtifactChunkRecords').mockImplementation(async (run, ...args) => {
            const scores = run.scoreList
            Object.defineProperty(run, 'scoreList', {get: () => scores})
            return await encodeChunk(run, ...args)
        })
        const generic = createStore()
        await encode(generic.store, {...chunk})
        await encode(candidate.store, chunk)
        assert.deepStrictEqual(candidate.writes, generic.writes)
        expect(candidate.copies).toStrictEqual([1, 1])
    })

    test('preserves origins through content-dictionary partitioning before offset splitting', async () => {
        const chunk = fixture()
        /**
         * @param {Fixture} input
         * @returns {Promise<ReturnType<typeof createStore>>}
         */
        const append = async (input) => {
            const result = createStore()
            Reflect.set(result.store, '_importSessionActive', true)
            vi.spyOn(result.store, '_ensureNextIdReadyForAppend').mockResolvedValue()
            vi.spyOn(result.store, '_getOrCreateShardState').mockResolvedValue(
                /** @type {NonNullable<Awaited<ReturnType<TermRecordOpfsStore['_getOrCreateShardState']>>>} */ ({}),
            )
            await result.store.appendBatchFromArtifactChunkResolvedContent(
                input,
                new Float64Array(0),
                new Uint32Array(0),
                Array.from({length: input.rowCount}, (_, i) => (i < 128 ? 'raw' : 'jmdict')),
            )
            return result
        }
        const generic = await append({...chunk})
        const candidate = await append(chunk)
        assert.deepStrictEqual(candidate.writes, generic.writes)
        expect(generic.copies).toStrictEqual([1, 1])
        expect(candidate.copies).toStrictEqual([0, 0])
    })
})
