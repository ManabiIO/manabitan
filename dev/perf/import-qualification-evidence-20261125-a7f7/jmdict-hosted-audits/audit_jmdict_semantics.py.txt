#!/usr/bin/env python3
"""Independently audit retained JMdict observations; never acquire browser runs.

Usage: python audit_jmdict_semantics.py --evidence DIR --repo REPO --node NODE
The repository must contain the three frozen commits. Original artifact ZIPs,
GitHub artifact metadata, and extracted AB/BA directories must be retained.
"""

import argparse
import datetime
import hashlib
import json
import pathlib
import re
import subprocess
import sys
import types
import zipfile


BASE = '7d1643e6147a10cb2165e9744cf0e63b59e46632'
CANDIDATE = '475d351ff07c1b5cfc5651d7f59c624629e4fdc5'
TOOLING = 'b1a0c5b78295def8cc91e4fca5e831d2248c7d01'
RUN = 45646399468
FLAGS = {'experimentalParserWorkers3': False}
CHANGED_ENTRY = 'js/dictionary/term-record-opfs-store.js'
HARNESS = 'test/chromium/extension-two-dictionary-import.e2e.js'
HELPERS = 'dev/perf/import-memory-20261125-a7f7/'


def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def json_object(pairs):
    result = {}
    for key, value in pairs:
        require(key not in result, f'Duplicate JSON object key: {key}')
        result[key] = value
    return result


def parse(data):
    return json.loads(data, object_pairs_hook=json_object,
                      parse_constant=lambda value: (_ for _ in ()).throw(ValueError(value)))


def read(path):
    return parse(path.read_text())


def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()


def equal(left, right, message):
    if isinstance(left, (bytes, set)) or isinstance(right, (bytes, set)):
        require(type(left) is type(right) and left == right, message)
    else:
        require(encoded(left) == encoded(right), message)


def git(repo, *args):
    return subprocess.check_output(['git', '-C', str(repo), *args])


def tree(repo, commit):
    result = {}
    for entry in git(repo, 'ls-tree', '-r', '-z', commit).split(b'\0'):
        if not entry:
            continue
        metadata, name = entry.split(b'\t', 1)
        mode, kind, oid = metadata.decode().split(' ')
        result[name.decode()] = {'mode': mode, 'kind': kind, 'oid': oid}
    return result


def blob(repo, commit, path):
    return git(repo, 'show', f'{commit}:{path}')


def module_from_commit(repo, name, relative):
    source = blob(repo, TOOLING, relative)
    module = types.ModuleType(name)
    exec(compile(source, f'{TOOLING}:{relative}', 'exec'), module.__dict__)
    return module, sha(source)


def zip_entries(path):
    with zipfile.ZipFile(path) as archive:
        infos = archive.infolist()
        names = [info.filename for info in infos]
        require(len(names) == len(set(names)), f'Duplicate ZIP members: {path}')
        for name in names:
            part = pathlib.PurePosixPath(name)
            require(not part.is_absolute() and '..' not in part.parts, f'Unsafe ZIP member: {name}')
        return {info.filename: sha(archive.read(info)) for info in infos if not info.is_dir()}


def timing_schedule(first):
    rows = []
    for number, order in enumerate((first, first[::-1]), 1):
        rows.extend({'id': f'warmup-{number}-{arm}', 'kind': 'warmup', 'pair': number, 'arm': arm}
                    for arm in order)
    for number in range(1, 13):
        order = first if number % 2 else first[::-1]
        rows.extend({'id': f'pair-{number:02d}-{arm}', 'kind': 'candidate', 'pair': number, 'arm': arm}
                    for arm in order)
        if number % 3 == 0:
            index = number // 3
            order = 'AC' if (index + (first == 'BA')) % 2 else 'CA'
            rows.extend({'id': f'control-{index:02d}-{arm}', 'kind': 'control', 'pair': index, 'arm': arm}
                        for arm in order)
    return rows


def collect_artifacts(root):
    result = {}
    for path in sorted(root.glob('artifacts*.json')):
        value = read(path)
        for item in value.get('artifacts', []):
            if item['name'] not in ('jmdict-two-workers-AB-a7f7', 'jmdict-two-workers-BA-a7f7'):
                continue
            if item['name'] in result:
                previous = result[item['name']]
                for key in ('id', 'digest', 'size_in_bytes', 'workflow_run'):
                    equal(item[key], previous[key], f'Conflicting artifact metadata: {item["name"]} {key}')
            result[item['name']] = item
    return result


def audit(root, repo, node, orders):
    receipts, receipts_hash = module_from_commit(repo, 'frozen_receipts', HELPERS + 'qualification_receipts.py')
    protocol, protocol_hash = module_from_commit(repo, 'frozen_protocol', HELPERS + 'protocol.py')
    commits = {'A': BASE, 'B': CANDIDATE, 'tooling': TOOLING}
    trees = {label: tree(repo, commit) for label, commit in commits.items()}
    required_same = {'package.json', 'package-lock.json', '.github/actions/setup/action.yml',
                     'test/perf/dictionaries.lock.json', 'dev/perf/import-benchmark.js',
                     'dev/perf/benchmark-support.js', 'dev/perf/dictionary-fixtures.js',
                     HARNESS, 'test/e2e/import-timing.js', 'dev/perf/import-media-validation.js'}
    required_same.update(name for name in trees['A']
                         if name.startswith(('test/e2e/', 'test/chromium/', 'dev/perf/'))
                         and name.endswith(('.js', '.mjs', '.html')))
    for name in required_same:
        equal(trees['A'][name], trees['B'].get(name), f'Candidate altered frozen harness/dependency: {name}')
        equal(trees['A'][name], trees['tooling'].get(name), f'Tooling altered frozen harness/dependency: {name}')
    runtime_delta = sorted(name for name in set(trees['A']) | set(trees['B'])
                           if name.startswith('ext/') and trees['A'].get(name) != trees['B'].get(name))
    equal(runtime_delta, ['ext/' + CHANGED_ENTRY], 'Unexpected runtime source delta')
    lock = parse(blob(repo, BASE, 'test/perf/dictionaries.lock.json'))
    fixture = lock['dictionaries']['jmdict']
    fixture_path = repo / 'builds/e2e-dictionary-cache' / fixture['cacheFile']
    if not fixture_path.exists():
        fixture_path = repo.parent / 'manabitan/builds/e2e-dictionary-cache' / fixture['cacheFile']
    fixture_bytes = fixture_path.read_bytes()
    equal(len(fixture_bytes), fixture['sizeBytes'], 'Local pinned fixture size mismatch')
    equal(sha(fixture_bytes), fixture['sha256'], 'Local pinned fixture digest mismatch')
    del fixture_bytes
    artifacts = collect_artifacts(root)
    reference = None
    media_reference = None
    expected_browser = None
    expected_entries = {}
    cohort_results = []
    native_inputs = []
    file_hashes = {}

    for order in orders:
        folder = root / order
        artifact = artifacts[f'jmdict-two-workers-{order}-a7f7']
        equal(artifact['workflow_run']['id'], RUN, 'Wrong source workflow run')
        equal(artifact['workflow_run']['head_sha'], TOOLING, 'Wrong qualification tooling commit')
        archive_path = root / (artifact['name'] + '.zip')
        equal(archive_path.stat().st_size, artifact['size_in_bytes'], 'Artifact size differs from GitHub metadata')
        archive_hash = sha(archive_path.read_bytes())
        equal('sha256:' + archive_hash, artifact['digest'], 'Artifact SHA differs from GitHub metadata')
        original_members = zip_entries(archive_path)
        for name, digest in original_members.items():
            path = folder / name
            require(path.is_file(), f'Missing extracted artifact member: {order}/{name}')
            equal(sha(path.read_bytes()), digest, f'Extracted artifact member changed: {order}/{name}')
            file_hashes[f'{order}/{name}'] = digest
        extracted_members = {str(p.relative_to(folder)) for p in folder.rglob('*') if p.is_file()}
        equal(sorted(extracted_members), sorted(original_members), f'Unrecorded extracted artifact file: {order}')

        plan = read(folder / 'registered-plan.json')
        for key, expected in [('baseline', BASE), ('candidate', CANDIDATE), ('dictionary', 'jmdict'),
                              ('initialOrder', order), ('pairs', 12), ('flags', FLAGS),
                              ('expectedWorkers', 2), ('expectedPackageEntries', [CHANGED_ENTRY])]:
            equal(plan[key], expected, f'{order}: plan {key}')
        expected_schedule = timing_schedule(order)
        equal(plan['schedule'], expected_schedule, f'{order}: altered timing schedule')
        equal(plan['memorySchedule'], protocol.schedule(1 if order == 'AB' else 2), f'{order}: memory schedule')
        identities = read(folder / 'source-identities.json')
        equal(identities['changedPackageEntries'], [CHANGED_ENTRY], 'Unexpected packaged source changes')
        equal(set(identities['arms']), {'A', 'B'}, 'Unexpected source arms')
        packages = {}
        for arm, commit in [('A', BASE), ('B', CANDIDATE)]:
            identity = identities['arms'][arm]
            equal(identity['sha'], commit, f'{order}/{arm}: source commit')
            equal(identity['tree'], git(repo, 'rev-parse', f'{commit}^{{tree}}').decode().strip(), f'{order}/{arm}: source tree')
            package = folder / f'package-{arm}.zip'
            package_hash = sha(package.read_bytes())
            equal(package_hash, identity['packageSha256'], f'{order}/{arm}: package bytes')
            entries = zip_entries(package)
            equal(entries, identity['entries'], f'{order}/{arm}: actual entry bytes differ from manifest')
            equal(len(entries), 551, f'{order}/{arm}: package entry count')
            with zipfile.ZipFile(package) as zipped:
                equal(sha(zipped.read(CHANGED_ENTRY)), sha(blob(repo, commit, 'ext/' + CHANGED_ENTRY)),
                      f'{order}/{arm}: packaged change differs from frozen source')
            if arm in expected_entries:
                equal(entries, expected_entries[arm], f'{order}/{arm}: cross-host package entry mismatch')
            else:
                expected_entries[arm] = entries
            equal(identity['lockSha256'], sha(blob(repo, commit, 'package-lock.json')), 'Dependency lock mismatch')
            equal(identity['fixtureLockSha256'], sha(blob(repo, commit, 'test/perf/dictionaries.lock.json')), 'Fixture lock mismatch')
            for name, digest in identity['sourceHashes'].items():
                if name == 'builds/manabitan-chrome.zip':
                    equal(digest, package_hash, 'Source receipt package mismatch')
                elif name == 'builds/manabitan-chrome-dev.zip':
                    equal(digest, None, 'Unexpected development package')
                else:
                    equal(digest, sha(blob(repo, commit, name)), f'{order}/{arm}: provenance file {name}')
            packages[arm] = {'sha256': package_hash, 'entries': len(entries),
                             'wasm': {name: digest for name, digest in entries.items() if name.endswith('.wasm')}}
        equal(set(identities['arms']['A']['entries']), set(identities['arms']['B']['entries']), 'Package inventory drift')
        package_delta = sorted(name for name in identities['arms']['A']['entries']
                               if identities['arms']['A']['entries'][name] != identities['arms']['B']['entries'][name])
        equal(package_delta, [CHANGED_ENTRY], 'Actual product package delta is not exactly allowed source file')
        browser = read(folder / 'browser-identity.json')
        equal(browser['version'], '153.0.8010.12', 'Unexpected Chromium version')
        require(re.fullmatch('[a-f0-9]{64}', browser['sha256']) is not None, 'Missing browser binary hash')
        if expected_browser is None:
            expected_browser = browser
        else:
            equal(browser, expected_browser, 'Cross-host browser binary identity mismatch')

        observed = read(folder / 'observations.json')
        equal(len(observed), 36, 'Missing or extra timing/warmup/control observation')
        equal([{key: row[key] for key in ('id', 'kind', 'pair', 'arm')} for row in observed],
              expected_schedule, 'Observed schedule differs from registered schedule')
        memory_plan = read(folder / 'memory/plan.json')
        equal(memory_plan['schedule'], plan['memorySchedule'], 'Memory plan mismatch')
        equal(memory_plan['sourceHashes'], identities['arms'], 'Memory source manifest mismatch')
        equal(memory_plan['authoritativeTiming'], False, 'Memory import timing incorrectly authoritative')
        original = (folder / 'memory/harness-original.js').read_bytes()
        overlay = (folder / 'memory/harness-instrumented.js').read_bytes()
        equal(original, blob(repo, BASE, HARNESS), 'Original memory harness differs from source')
        equal(overlay, protocol.memory_overlay(original.decode()).encode(), 'Memory overlay exceeds registered marker-only patch')
        equal(sha(overlay), memory_plan['overlaySha256'], 'Memory overlay hash mismatch')
        equal(read(folder / 'memory/semantic-failures.json'), [], 'Recorded memory semantic failure')
        memory_observations = read(folder / 'memory/observations.json')
        equal([{key: row[key] for key in ('name', 'arm', 'index', 'pair')} for row in memory_observations],
              plan['memorySchedule'], 'Observed memory schedule differs')
        items = [{**row, 'mode': 'sample', 'memory': False} for row in observed]
        items += [{'id': f'exhaustive-media-{arm}', 'arm': arm, 'kind': 'exhaustive-media', 'mode': 'all', 'memory': False}
                  for arm in order]
        items += [{'id': 'memory/' + row['name'], 'arm': row['arm'], 'kind': 'memory', 'mode': 'sample', 'memory': True}
                  for row in plan['memorySchedule']]
        equal(sorted(str(p.parent.relative_to(folder)) for p in folder.rglob('summary.json')),
              sorted(row['id'] for row in items), 'Unexpected or missing import summary')
        equal(sorted(str(p.parent.relative_to(folder)) for p in folder.rglob('run-1.json')),
              sorted(row['id'] for row in items), 'Unexpected or missing raw report')
        observations = []
        last_started = None
        host = None
        for item in items:
            output = folder / item['id']
            summary = read(output / 'summary.json')
            report = read(output / 'run-1.json')
            arm = 'B' if item['arm'] == 'B' else 'A'
            identity = identities['arms'][arm]
            hashes = identity['sourceHashes'].copy()
            if item['memory']:
                hashes[HARNESS] = sha(overlay)
            canonical = receipts.validate_import_summary(
                summary, report, dictionary='jmdict', fixture=fixture, source_sha=identity['sha'],
                package_sha256=identity['packageSha256'], browser_version=browser['version'],
                import_flags=FLAGS, source_hashes=hashes, source_dirty=item['memory'], media_check=item['mode'])
            equal(canonical, read(output / 'canonical-receipt.json'), 'Stored canonical receipt differs from raw reconstruction')
            if reference is None:
                reference = canonical
            else:
                receipts.require_matching_receipts(reference, canonical, compare_media=item['mode'] == 'sample')
            if item['mode'] == 'all':
                if media_reference is None:
                    media_reference = canonical
                else:
                    receipts.require_matching_receipts(media_reference, canonical)
            equal(report.get('failureReason'), '', 'Report failure present')
            equal(report.get('skipReason'), '', 'Report skip present')
            equal(report['phaseCount'], len(report['phases']), 'Raw phase count mismatch')
            run = summary['runs'][0]
            if 'totalImportMs' in item:
                equal(run['totalImportMs'], item['totalImportMs'], 'Recorded observation duration differs from raw report')
            breakdown = run['step4Breakdown']['dictionaries']
            equal(len(breakdown), 1, 'Unexpected multiple imported dictionaries')
            equal(breakdown[0]['parserWorkerCount'], 2, 'Expected two parser workers did not execute')
            phases = run['importDebug']['importerPhaseTimings']
            parsed = [phase['details'] for phase in phases if phase['phase'].startswith('term-file-fast-path:')]
            equal(len(parsed), 1, 'Missing or duplicate fast-path import')
            details = parsed[0]
            equal(details['parserParallelWorkerCount'], 2, 'Raw worker policy differs from summary')
            equal(details['parserExperiments']['experimentalParserWorkers3'], False, 'Effective worker flag changed')
            equal(details['rows'], fixture['termRows'], 'Fast parser rows differ from corpus')
            equal(details['parserParallelGroupCount'], breakdown[0]['parserGroupCount'], 'Raw group count differs from summary')
            archive = [phase['details'] for phase in phases if phase['phase'] == 'archive-and-index']
            equal(len(archive), 1, 'Unexpected archive phase count')
            equal(archive[0]['archiveSizeBytes'], fixture['sizeBytes'], 'Actual archive source size mismatch')
            started = datetime.datetime.fromisoformat(report['startedAtIso'].replace('Z', '+00:00'))
            if last_started is not None:
                require(started > last_started, 'Recorded acquisition chronology differs from fixed schedule')
            last_started = started
            current_host = {key: summary['hostEnvironment'][key] for key in
                            ('nodeVersion', 'v8Version', 'platform', 'arch', 'cpuModel', 'logicalCpuCount',
                             'availableParallelism', 'totalMemoryBytes')}
            if host is None:
                host = current_host
            else:
                equal(current_host, host, 'Host hardware/runtime changed within cohort')
            native_inputs.append({'directory': str(output), 'mode': item['mode']})
            observations.append({
                'id': item['id'], 'arm': item['arm'], 'kind': item['kind'], 'startedAtIso': report['startedAtIso'],
                'source': identity['sha'], 'packageSha256': identity['packageSha256'],
                'summarySha256': sha((output / 'summary.json').read_bytes()),
                'rawReportSha256': sha((output / 'run-1.json').read_bytes()),
                'canonicalReceiptSha256': sha(encoded(canonical)),
                'sourceDirty': summary['source']['dirty'], 'traceEnabled': report['benchmark']['traceEnabled'],
                'builtInProcessSampling': report['benchmark']['processSampling'],
                'externalMemorySampling': item['memory'],
                'parserWorkers': details['parserParallelWorkerCount'], 'parserGroups': details['parserParallelGroupCount'],
                'parserNativeSegments': details['parserNativeSegmentedLookupSegments'],
                'termRows': canonical['common']['validation']['termRows'],
                'outputDigestSha256': canonical['common']['validation']['outputDigestSha256'],
                'lookupReceiptSha256': sha(encoded(canonical['common']['validation']['lookup'])),
                'storage': canonical['common']['validation']['storage'], 'media': canonical['media'],
                'semanticsPassed': True,
            })
        cohort_results.append({'order': order, 'artifact': {'id': artifact['id'], 'bytes': archive_path.stat().st_size,
                              'sha256': archive_hash, 'fileCount': len(original_members)},
                               'host': host, 'packages': packages, 'changedPackageEntries': package_delta,
                               'observations': observations, 'semanticsPassed': True})

    # Re-execute the unchanged production extraction validator on every raw
    # report. This is pure report parsing; it launches no browser or import.
    dependencies = ('dev/perf/benchmark-support.js', 'ext/js/core/json.js',
                    'ext/js/dictionary/term-bank-experiments.js')
    for name in dependencies:
        equal((repo / name).read_bytes(), blob(repo, BASE, name), 'Local native-validator dependency changed')
    native_script = r'''
import {readFileSync} from 'node:fs'
import {pathToFileURL} from 'node:url'
import {isDeepStrictEqual} from 'node:util'
const {extractImportResult} = await import(pathToFileURL(process.argv[1]).href)
const input = JSON.parse(readFileSync(0, 'utf8'))
const results = []
for (const row of input.observations) {
    const report = JSON.parse(readFileSync(row.directory + '/run-1.json', 'utf8'))
    const summary = JSON.parse(readFileSync(row.directory + '/summary.json', 'utf8'))
    const actual = extractImportResult(report, 'jmdict', input.fixture, false, input.flags, row.mode)
    for (const [key, value] of Object.entries(actual)) {
        if (!isDeepStrictEqual(value, summary.runs[0][key])) {
            throw new Error(`${row.directory}: native extraction mismatch for ${key}`)
        }
    }
    results.push(row.directory)
}
process.stdout.write(JSON.stringify({count: results.length, passed: true}))
'''
    native = subprocess.check_output([node, '--input-type=module', '--eval', native_script,
                                      str(repo / 'dev/perf/benchmark-support.js')],
                                     input=encoded({'observations': native_inputs, 'fixture': fixture, 'flags': FLAGS}))
    native_result = parse(native)
    equal(native_result, {'count': 42 * len(orders), 'passed': True}, 'Native extraction validator incomplete')
    return {
        'schemaVersion': 1, 'status': 'pass' if len(orders) == 2 else 'partial-single-host-pass',
        'scope': 'Independent source, actual package, raw report, canonical semantic, storage, restart, lookup, media and two-worker-policy audit; no new observations acquired.',
        'run': RUN, 'runUrl': f'https://github.com/ManabiIO/manabitan/actions/runs/{RUN}',
        'sourceCommits': commits, 'runtimeChangedFiles': runtime_delta,
        'frozenHarnessDependencyFiles': len(required_same), 'fixture': fixture,
        'browser': expected_browser, 'observationsAudited': len(native_inputs),
        'canonicalCommonSha256': sha(encoded(reference['common'])),
        'sampleMediaReceipt': reference['media'], 'fullMediaReceipt': media_reference['media'],
        'nativeProductionExtractionAudit': native_result,
        'helperHashes': {'qualification_receipts.py': receipts_hash, 'protocol.py': protocol_hash},
        'auditScriptSha256': sha(pathlib.Path(__file__).read_bytes()),
        'cohorts': cohort_results, 'rawArtifactFileHashes': file_hashes,
        'limitations': [
            'Storage equality uses exact logical per-file sizes, with only validated random SAH basenames normalized.',
            'The fixture hash is independently verified against retained local pinned source bytes; hosted reports bind the same lock, size and revision.',
            'Browser binary SHA is a retained host receipt; the browser executable itself is not archived here.',
            'Output and lookup digests are compared exactly as recorded by the unchanged production verifier; persisted browser profile files are not retained.',
            'JMdict contains zero media files; the exhaustive-media validation confirms that exact empty inventory.',
            'Sampled import durations are excluded from performance interpretation; a separate auditor owns raw process-memory coverage and gates.',
            'This semantic audit does not establish a timing improvement; each host remains a separate cohort.',
        ],
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--evidence', required=True, type=pathlib.Path)
    parser.add_argument('--repo', required=True, type=pathlib.Path)
    parser.add_argument('--node', required=True)
    parser.add_argument('--orders', nargs='+', choices=('AB', 'BA'), default=['AB', 'BA'])
    args = parser.parse_args()
    require(len(args.orders) == len(set(args.orders)), 'Repeated cohort')
    result = audit(args.evidence.resolve(), args.repo.resolve(), args.node, args.orders)
    output = args.evidence / ('independent-semantic-audit.json' if len(args.orders) == 2 else 'independent-semantic-audit-partial.json')
    output.write_text(json.dumps(result, indent=2, sort_keys=True) + '\n')
    print(json.dumps({'status': result['status'], 'observations': result['observationsAudited'],
                      'output': str(output), 'sha256': sha(output.read_bytes())}))


if __name__ == '__main__':
    main()
