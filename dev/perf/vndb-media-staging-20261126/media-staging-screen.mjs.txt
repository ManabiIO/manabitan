const modulePath = 'ext/js/dictionary/term-bank-wasm-parser.js'
const wasmPath = 'ext/lib/term-bank-parser.wasm'
const hash = (value) => createHash('sha256').update(value).digest('hex')
const baselineCode = await readFile(path.join(baselineRoot, modulePath))
const candidateCode = await readFile(path.join(candidateRoot, modulePath))

assert.equal(hash(native), hash(await readFile(path.join(candidateRoot, wasmPath))))

assert.equal((await stat(fixturePath)).size, lock.sizeBytes)

const {RawZipPayloadReader, RAW_ZIP_BATCH_MAX_FILES, RAW_ZIP_BATCH_MAX_PAYLOAD_BYTES} = await import(pathToFileURL(path.join(baselineRoot, 'ext/js/dictionary/term-bank-source-pipeline.js')))
const {getImageMediaTypeFromFileName} = await import(pathToFileURL(path.join(baselineRoot, 'ext/js/media/media-util.js')))

async function prepare() {
    const archive = await openAsBlob(fixturePath)
    const entries = await readDenseZipDirectory(archive)
    assert(entries)
    const files = entries.filter(({filename}) => getImageMediaTypeFromFileName(filename) !== null)
    assert.equal(files.length, lock.mediaFiles)
    const reader = new RawZipPayloadReader(archive)
    const batches = []
    let rangeReadCount = 0
    let rangeReadBytes = 0
    for (let start = 0; start < files.length;) {
        let end = start
        let compressed = 0
        let decoded = 0
        while (end < files.length && end - start < RAW_ZIP_BATCH_MAX_FILES) {
            const file = files[end]
            if (compressed + file.compressedSize > RAW_ZIP_BATCH_MAX_PAYLOAD_BYTES || decoded + file.uncompressedSize > RAW_ZIP_BATCH_MAX_PAYLOAD_BYTES) break
            compressed += file.compressedSize
            decoded += file.uncompressedSize
            ++end
        }
        assert(end > start)
        const result = await reader.readBatch(files.slice(start, end), undefined, () => false, true)
        assert(result)
        rangeReadCount += result.rangeReadCount
        rangeReadBytes += result.rangeReadBytes
        batches.push(result.sources)
        start = end
    }
    return {batches, rangeReadCount, rangeReadBytes}
}

const {batches, rangeReadCount, rangeReadBytes} = await prepare()
const mediaFiles = batches.reduce((sum, sources) => sum + sources.length, 0)
const mediaBytes = batches.reduce((sum, sources) => sum + sources.reduce((s, source) => s + source.uncompressedSize, 0), 0)
assert.equal(mediaFiles, lock.mediaFiles)
assert.equal(mediaBytes, lock.mediaBytes)

function sourceDigest() {
    const digest = createHash('sha256')
    for (const sources of batches) {
        for (const source of sources) {
            digest.update(JSON.stringify([source.filename, source.compressionMethod, source.compressedSize, source.uncompressedSize, source.signature]) + '\n')
            digest.update(source.bytes)
        }
    }
    return digest.digest('hex')
}

const beforeDigest = sourceDigest()
const sourceBuffers = new Set(batches.flatMap((sources) => sources.map(({bytes}) => bytes.buffer)))
const work = {}
for (const arm of ['A', 'B', 'C']) {
    const counts = {copies: 0, copiedBytes: 0, maxCopiedSpanBytes: 0, maxDestinationBufferBytes: 0}
    const original = Uint8Array.prototype.set
    Uint8Array.prototype.set = function (source, offset) {
        if (ArrayBuffer.isView(source) && sourceBuffers.has(source.buffer)) {
            ++counts.copies
            counts.copiedBytes += source.byteLength
            counts.maxCopiedSpanBytes = Math.max(counts.maxCopiedSpanBytes, source.byteLength)
            counts.maxDestinationBufferBytes = Math.max(counts.maxDestinationBufferBytes, this.buffer.byteLength)
        }
        return Reflect.apply(original, this, [source, offset])
    }
    try {
        for (const sources of batches) await modules[arm].validateCompressedZipMediaBatchWasm(sources)
    } finally { Uint8Array.prototype.set = original }
    work[arm] = counts
    await writeFile(outputPath + '.work-checkpoint', JSON.stringify({work, fixtureSha256: lock.sha256, sourcesSha256: beforeDigest}) + '\n')
}

const observations = []
async function run(arm, phase, round, order) {
    globalThis.gc?.()
    const loadBefore = loadavg()
    const usageBefore = process.cpuUsage()
    const start = performance.now()
    for (const sources of batches) await modules[arm].validateCompressedZipMediaBatchWasm(sources)
    const wallMs = performance.now() - start
    const cpu = process.cpuUsage(usageBefore)
    observations.push({arm, phase, round, order, wallMs, processCpuMs: (cpu.user + cpu.system) / 1000, loadBefore, loadAfter: loadavg()})
    process.stderr.write(JSON.stringify(observations.at(-1)) + '\n')
}
if (mode === 'all') {
    for (const [round, order] of ['ABC', 'CBA'].entries()) for (const arm of order) await run(arm, 'warmup', round, order)
    for (const [round, order] of ['ABC', 'BCA', 'CAB', 'CBA', 'ACB', 'BAC'].entries()) for (const arm of order) await run(arm, 'measured', round, order)
}
assert.equal(sourceDigest(), beforeDigest)

batches.length = 0

async function packedOutput(root) {
    const {preparePackedSourceZipMediaEntries} = await import(pathToFileURL(path.join(root, 'ext/js/dictionary/source-zip-media-pack.js')))
    const {createGenerationMediaIndex} = await import(pathToFileURL(path.join(root, 'ext/js/dictionary/dictionary-generation-media-store.js')))
    const parser = await import(pathToFileURL(path.join(root, modulePath)))
    parser.setTermBankWasmModule(compiled)
    const archive = await openAsBlob(fixturePath)
    const entries = await readDenseZipDirectory(archive)
    assert(entries)
    const files = entries.filter(({filename}) => getImageMediaTypeFromFileName(filename) !== null)
    const packed = await preparePackedSourceZipMediaEntries(archive, files, new RawZipPayloadReader(archive), () => {})
    assert(packed)
    assert.equal(packed.records.length, lock.mediaFiles)
    assert.equal(packed.uncompressedBytes, lock.mediaBytes)
    const index = createGenerationMediaIndex(packed.records, packed.blob.size)
    assert(index)
    const recordsHash = createHash('sha256')
    for (const record of packed.records) {
        recordsHash.update(JSON.stringify([
            record.path, record.mediaType, record.offset, record.compressedSize,
            record.packedOffset, record.packedLength, record.uncompressedLength,
            record.compressionMethod, record.signature,
        ]) + '\n')
    }
    const blobHash = createHash('sha256')
    for await (const bytes of packed.blob.stream()) blobHash.update(bytes)
    return {
        records: packed.records.length,
        recordsSha256: recordsHash.digest('hex'),
        indexSha256: hash(index.index),
        descriptor: index.descriptor,
        packedSpanBytes: packed.blob.size,
        packedSpanSha256: blobHash.digest('hex'),
        rangeReadCount: packed.rangeReadCount,
        rangeReadBytes: packed.rangeReadBytes,
        compressedBytes: packed.compressedBytes,
        uncompressedBytes: packed.uncompressedBytes,
    }
}
