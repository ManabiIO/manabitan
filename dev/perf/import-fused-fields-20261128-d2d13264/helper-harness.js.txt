const U32_NULL = 0xffffffff;
const MAX_CONTENT_OFFSET_DELTA = U32_NULL - 1;
const U16_NULL = 0xffff;
const FLOAT64_SCORE_RECORD_FIELDS_BYTES = 16;
const LOOKUP_INDEX_RECORD_FIELDS_BYTES = 12;
const COMPACT_RECORD_FIELDS_HEADER_BYTES = 16;
const COMPACT_RECORD_FIELDS_BYTES_PER_ROW = 8;
const COMPACT_RECORD_FIELDS_MAGIC = 0x3246524d;
const LOOKUP_INDEX_RECORD_FIELDS_FORMAT_COMPACT = 2;
const COMPACT_SCORE_TABLE_MIN_ROWS = 64;
const COMPACT_SCORE_TABLE_RADIUS = 256;
const resolvedContentReferenceOrigins = new WeakMap();
const ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR = Object.getOwnPropertyDescriptor(ArrayBuffer.prototype, 'byteLength');
function hasFixedContentSpan(chunk, count) {
    return (
        count > 0 &&
        typeof chunk.fixedContentOffsetBase === 'number' &&
        Number.isFinite(chunk.fixedContentOffsetBase) &&
        chunk.fixedContentOffsetBase >= 0 &&
        typeof chunk.fixedContentLength === 'number' &&
        Number.isFinite(chunk.fixedContentLength) &&
        chunk.fixedContentLength >= 0
    );
}

function getArtifactContentOffset(chunk, contentOffsets, index, useFixedContentSpan = hasFixedContentSpan(chunk, index + 1)) {
    if (useFixedContentSpan) {
        return /** @type {number} */ (chunk.fixedContentOffsetBase) + (index * /** @type {number} */ (chunk.fixedContentLength));
    }
    const references = chunk.resolvedContentReferences;
    if (typeof references === 'object' && references !== null) {
        const uniqueIndex = references.uniqueIndexList[index];
        if (uniqueIndex >= references.offsets.length) {
            throw new RangeError(`Invalid resolved term content reference index: ${uniqueIndex}`);
        }
        return references.offsets[uniqueIndex];
    }
    return contentOffsets[index];
}

function getArtifactContentLength(chunk, contentLengths, index) {
    if (typeof chunk.fixedContentLength === 'number') { return chunk.fixedContentLength; }
    const references = chunk.resolvedContentReferences;
    if (typeof references === 'object' && references !== null) {
        const uniqueIndex = references.uniqueIndexList[index];
        if (uniqueIndex >= references.lengths.length) {
            throw new RangeError(`Invalid resolved term content reference index: ${uniqueIndex}`);
        }
        return references.lengths[uniqueIndex];
    }
    return contentLengths[index];
}

function isLosslessInt32Score(score) {
    return (
        Number.isInteger(score) &&
        score >= -0x80000000 &&
        score <= 0x7fffffff &&
        !Object.is(score, -0)
    );
}

function validateContentOffset(value) {
    if (value !== -1 && (!Number.isSafeInteger(value) || value < 0)) {
        throw new RangeError(`Invalid term content offset: ${value}`);
    }
}

function validateContentLength(value) {
    if (value !== -1 && (!Number.isSafeInteger(value) || value < 0 || value >= U32_NULL)) {
        throw new RangeError(`Invalid term content length: ${value}`);
    }
}

function getSynchronousResolvedContentReferences(chunk, count) {
    const origin = resolvedContentReferenceOrigins.get(chunk);
    if (
        typeof origin === 'undefined' || Object.getPrototypeOf(chunk) !== Object.prototype ||
        typeof ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR?.get !== 'function'
    ) { return null; }
    const fixedLengthDescriptor = Object.getOwnPropertyDescriptor(chunk, 'fixedContentLength');
    if (
        typeof fixedLengthDescriptor === 'undefined' ?
            'fixedContentLength' in chunk :
            !('value' in fixedLengthDescriptor) || typeof fixedLengthDescriptor.value === 'number'
    ) { return null; }
    const references = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(chunk, 'resolvedContentReferences')?.value);
    if (references !== origin || Object.getPrototypeOf(references) !== Object.prototype) { return null; }
    const uniqueIndexList = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(references, 'uniqueIndexList')?.value);
    const offsets = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(references, 'offsets')?.value);
    const lengths = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(references, 'lengths')?.value);
    if (
        !(uniqueIndexList instanceof Uint32Array) ||
        !(offsets instanceof Float64Array) ||
        !(lengths instanceof Uint32Array) ||
        Object.getPrototypeOf(uniqueIndexList) !== Uint32Array.prototype ||
        Object.getPrototypeOf(offsets) !== Float64Array.prototype ||
        Object.getPrototypeOf(lengths) !== Uint32Array.prototype
    ) { return null; }
    for (const column of [uniqueIndexList, offsets, lengths]) {
        if (
            !ArrayBuffer.isView(column) ||
            Object.hasOwn(column, 'length') ||
            Object.hasOwn(column, 'buffer') ||
            !(column.buffer instanceof ArrayBuffer)
        ) { return null; }
        // The builtin getter establishes private ArrayBuffer storage using
        // the buffer's internal identity, independently of its prototype.
        try {
            ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR.get.call(column.buffer);
        } catch {
            return null;
        }
    }
    if (uniqueIndexList.length < count || offsets.length !== lengths.length) { return null; }
    return {uniqueIndexList, offsets, lengths};
}

function getSynchronousRecordFieldReferences(chunk, count) {
    const references = getSynchronousResolvedContentReferences(chunk, count)
    if (references === null) { return null }
    const scores = /** @type {unknown} */ (Object.getOwnPropertyDescriptor(chunk, 'scoreList')?.value)
    if (
        typeof ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR?.get !== 'function' ||
        !(scores instanceof Int32Array) || !ArrayBuffer.isView(scores) ||
        Object.getPrototypeOf(scores) !== Int32Array.prototype ||
        Object.hasOwn(scores, 'length') || Object.hasOwn(scores, 'buffer') ||
        scores.length < count
    ) { return null }
    try {
        ARRAY_BUFFER_BYTE_LENGTH_DESCRIPTOR.get.call(scores.buffer)
    } catch { return null }
    return references
}
function createArtifactTermRecordChunkWithResolvedContent(chunk, uniqueIndexList, offsets, lengths, preparedLookupIndexes = null) {
    const references = {uniqueIndexList, offsets, lengths};
    const result = {...chunk, ...(preparedLookupIndexes === null ? {} : {preparedLookupIndexes}), resolvedContentReferences: references};
    resolvedContentReferenceOrigins.set(result, references);
    return result;
}
/**
 * Check native view identity before any instanceof walk can visit a proxy
 * prototype. Generic, shared, shadowed or subclass columns stay on the
 * established validator and retain its property-read ordering.
 * @param {unknown} value
 * @param {object} prototype
 * @returns {boolean}
 */
function isPlainPrivateRecordFieldColumn(value, prototype) {
    return (
        ArrayBuffer.isView(value) && Object.getPrototypeOf(value) === prototype &&
        !Object.hasOwn(value, 'buffer') && !Object.hasOwn(value, 'length') &&
        Object.getPrototypeOf(value.buffer) === ArrayBuffer.prototype
    );
}

/**
 * Validate a whole compact run while assigning its score keys. Only factory
 * chunks with private, ordinary columns are admitted; no input is retained.
 * A null result leaves the original split/legacy path responsible for the run.
 * @param {{rowCount: number, scoreList: number[]|Int32Array|Float64Array, fixedContentOffsetBase?: number, fixedContentLength?: number, resolvedContentReferences?: ResolvedContentReferences}} chunk
 * @param {number} count
 * @returns {{recordFields: Uint8Array, contentOffsetBase: number}|null}
 * @throws {RangeError} If a selected content reference is invalid.
 */
function tryEncodeCompactRecordFieldsWithContentBounds(chunk, count) {
    if (!Number.isSafeInteger(count) || count <= 0) { return null; }
    const origin = resolvedContentReferenceOrigins.get(chunk);
    if (typeof origin === 'undefined') { return null; }
    const fixedBaseDescriptor = Object.getOwnPropertyDescriptor(chunk, 'fixedContentOffsetBase');
    if (
        Object.getPrototypeOf(chunk) !== Object.prototype ||
        (typeof fixedBaseDescriptor === 'undefined' ? 'fixedContentOffsetBase' in chunk : !('value' in fixedBaseDescriptor)) ||
        Object.getOwnPropertyDescriptor(chunk, 'rowCount')?.value !== count ||
        Object.getOwnPropertyDescriptor(chunk, 'resolvedContentReferences')?.value !== origin ||
        Object.getPrototypeOf(origin) !== Object.prototype ||
        !isPlainPrivateRecordFieldColumn(Object.getOwnPropertyDescriptor(origin, 'uniqueIndexList')?.value, Uint32Array.prototype) ||
        !isPlainPrivateRecordFieldColumn(Object.getOwnPropertyDescriptor(origin, 'offsets')?.value, Float64Array.prototype) ||
        !isPlainPrivateRecordFieldColumn(Object.getOwnPropertyDescriptor(origin, 'lengths')?.value, Uint32Array.prototype) ||
        !isPlainPrivateRecordFieldColumn(Object.getOwnPropertyDescriptor(chunk, 'scoreList')?.value, Int32Array.prototype)
    ) { return null; }
    const references = getSynchronousRecordFieldReferences(chunk, count);
    if (references === null) { return null; }
    const {uniqueIndexList, offsets, lengths} = references;
    const scoreList = /** @type {Int32Array} */ (chunk.scoreList);
    const scoreKeys = new Uint16Array(count);
    /** @type {number[]} */
    const scores = [];
    /** @type {Map<number, number>} */
    const scoreIndexes = new Map();
    const inlineScoreIndexes = count >= COMPACT_SCORE_TABLE_MIN_ROWS ?
        new Uint16Array(COMPACT_SCORE_TABLE_RADIUS * 2) :
        null;
    let minimumOffset = Number.POSITIVE_INFINITY;
    let maximumOffset = Number.NEGATIVE_INFINITY;
    let previousUnique = -1;
    for (let i = 0; i < count; ++i) {
        const unique = uniqueIndexList[i];
        if (unique !== previousUnique) {
            if (unique >= offsets.length) {
                throw new RangeError(`Invalid resolved term content reference index: ${unique}`);
            }
            const offset = offsets[unique];
            const length = lengths[unique];
            validateContentOffset(offset);
            validateContentLength(length);
            if (offset >= 0) {
                minimumOffset = Math.min(minimumOffset, offset);
                maximumOffset = Math.max(maximumOffset, offset);
            }
            if (maximumOffset - minimumOffset > MAX_CONTENT_OFFSET_DELTA || length >= U16_NULL) {
                return null;
            }
            previousUnique = unique;
        }
        const score = scoreList[i];
        const inlineScore = score + COMPACT_SCORE_TABLE_RADIUS;
        const useInlineScore = inlineScoreIndexes !== null && inlineScore >= 0 && inlineScore < inlineScoreIndexes.length;
        let scoreIndex = useInlineScore ? inlineScoreIndexes[inlineScore] - 1 : (scoreIndexes.get(score) ?? -1);
        if (scoreIndex < 0) {
            scoreIndex = scores.length;
            if (scoreIndex >= U16_NULL) { return null; }
            if (useInlineScore) {
                inlineScoreIndexes[inlineScore] = scoreIndex + 1;
            } else {
                scoreIndexes.set(score, scoreIndex);
            }
            scores.push(score);
        }
        scoreKeys[i] = scoreIndex;
    }
    const byteLength = COMPACT_RECORD_FIELDS_HEADER_BYTES +
    (scores.length * 4) +
    (count * COMPACT_RECORD_FIELDS_BYTES_PER_ROW);
    if (scores.length === 0 || scores.length >= U16_NULL || byteLength >= count * LOOKUP_INDEX_RECORD_FIELDS_BYTES) {
        return null;
    }
    const contentOffsetBase = minimumOffset === Number.POSITIVE_INFINITY ? 0 : minimumOffset;
    const recordFields = new Uint8Array(byteLength);
    const header = new Uint32Array(recordFields.buffer, 0, COMPACT_RECORD_FIELDS_HEADER_BYTES / 4);
    header[0] = COMPACT_RECORD_FIELDS_MAGIC;
    header[1] = count;
    header[2] = scores.length;
    header[3] = LOOKUP_INDEX_RECORD_FIELDS_FORMAT_COMPACT;
    let cursor = COMPACT_RECORD_FIELDS_HEADER_BYTES;
    new Int32Array(recordFields.buffer, cursor, scores.length).set(scores);
    cursor += scores.length * 4;
    const outputOffsets = new Uint32Array(recordFields.buffer, cursor, count);
    cursor += count * 4;
    const outputLengths = new Uint16Array(recordFields.buffer, cursor, count);
    for (let i = 0; i < count; ++i) {
        const unique = uniqueIndexList[i];
        const offset = offsets[unique];
        outputOffsets[i] = offset < 0 ? U32_NULL : offset - contentOffsetBase;
        outputLengths[i] = lengths[unique];
    }
    cursor += count * 2;
    new Uint16Array(recordFields.buffer, cursor, count).set(scoreKeys);
    return {recordFields, contentOffsetBase};
}

class Encoder {    _encodeArtifactRecordFields(chunk, contentOffsets, contentLengths, contentOffsetBase) {
        const count = chunk.rowCount;
        const useFixedContentSpan = hasFixedContentSpan(chunk, count);
        const useFloat64Scores = !(chunk.scoreList instanceof Int32Array) && chunk.scoreList.some((score) => !isLosslessInt32Score(score ?? 0));
        if (useFloat64Scores) {
            const output = new Uint8Array(count * FLOAT64_SCORE_RECORD_FIELDS_BYTES);
            const view = new DataView(output.buffer);
            for (let i = 0; i < count; ++i) {
                const fieldOffset = i * FLOAT64_SCORE_RECORD_FIELDS_BYTES;
                const contentOffset = getArtifactContentOffset(chunk, contentOffsets, i, useFixedContentSpan);
                const contentLength = getArtifactContentLength(chunk, contentLengths, i);
                view.setUint32(fieldOffset, contentOffset < 0 ? U32_NULL : contentOffset - contentOffsetBase, true);
                view.setUint32(fieldOffset + 4, contentLength < 0 ? U32_NULL : contentLength, true);
                view.setFloat64(fieldOffset + 8, chunk.scoreList[i] ?? 0, true);
            }
            return output;
        }
        /* eslint-disable @stylistic/semi */
        const directReferences = getSynchronousRecordFieldReferences(chunk, count)
        const offsets = directReferences === null ? new Uint32Array(count) : null
        const lengths = directReferences === null ? new Uint16Array(count) : null
        /* eslint-enable @stylistic/semi */
        const scoreKeys = new Uint16Array(count);
        /** @type {number[]} */
        const scores = [];
        /** @type {Map<number, number>} */
        const scoreIndexes = new Map();
        // Common integer scores use a bounded table. Store the key plus one
        // so zero can mark an unseen score without initializing the table.
        const inlineScoreIndexes = chunk.scoreList instanceof Int32Array && count >= COMPACT_SCORE_TABLE_MIN_ROWS ?
            new Uint16Array(COMPACT_SCORE_TABLE_RADIUS * 2) :
            null;
        let compact = true;
        for (let i = 0; i < count; ++i) {
            /* eslint-disable-next-line @stylistic/semi */
            const contentOffset = directReferences === null ? getArtifactContentOffset(chunk, contentOffsets, i, useFixedContentSpan) : 0
            const contentLength = getArtifactContentLength(chunk, contentLengths, i);
            if (contentLength >= U16_NULL) {
                compact = false;
                break;
            }
            /* eslint-disable @stylistic/semi */
            if (offsets !== null && lengths !== null) {
                offsets[i] = contentOffset < 0 ? U32_NULL : contentOffset - contentOffsetBase
                lengths[i] = contentLength < 0 ? U16_NULL : contentLength
            }
            /* eslint-enable @stylistic/semi */
            const score = chunk.scoreList[i] ?? 0;
            const inlineScore = score + COMPACT_SCORE_TABLE_RADIUS;
            const useInlineScore = inlineScoreIndexes !== null && inlineScore >= 0 && inlineScore < inlineScoreIndexes.length;
            let scoreIndex = useInlineScore ? inlineScoreIndexes[inlineScore] - 1 : (scoreIndexes.get(score) ?? -1);
            if (scoreIndex < 0) {
                scoreIndex = scores.length;
                if (scoreIndex >= U16_NULL) {
                    compact = false;
                    break;
                }
                if (useInlineScore) {
                    inlineScoreIndexes[inlineScore] = scoreIndex + 1;
                } else {
                    scoreIndexes.set(score, scoreIndex);
                }
                scores.push(score);
            }
            scoreKeys[i] = scoreIndex;
        }
        const compactByteLength = COMPACT_RECORD_FIELDS_HEADER_BYTES +
        (scores.length * 4) +
        (count * COMPACT_RECORD_FIELDS_BYTES_PER_ROW);
        if (
            !compact ||
            scores.length === 0 ||
            // The reader reserves this score count even though the last key fits u16.
            scores.length >= U16_NULL ||
            compactByteLength >= count * LOOKUP_INDEX_RECORD_FIELDS_BYTES
        ) {
            const legacy = new Uint8Array(count * LOOKUP_INDEX_RECORD_FIELDS_BYTES);
            const legacyU32 = new Uint32Array(legacy.buffer);
            const legacyI32 = new Int32Array(legacy.buffer);
            for (let i = 0; i < count; ++i) {
                const fieldOffset = i * 3;
                const contentOffset = getArtifactContentOffset(chunk, contentOffsets, i, useFixedContentSpan);
                const contentLength = getArtifactContentLength(chunk, contentLengths, i);
                legacyU32[fieldOffset] = contentOffset < 0 ? U32_NULL : contentOffset - contentOffsetBase;
                legacyU32[fieldOffset + 1] = contentLength < 0 ? U32_NULL : contentLength;
                legacyI32[fieldOffset + 2] = chunk.scoreList[i] ?? 0;
            }
            return legacy;
        }
        const output = new Uint8Array(compactByteLength);
        const header = new Uint32Array(output.buffer, 0, COMPACT_RECORD_FIELDS_HEADER_BYTES / 4);
        header[0] = COMPACT_RECORD_FIELDS_MAGIC;
        header[1] = count;
        header[2] = scores.length;
        header[3] = LOOKUP_INDEX_RECORD_FIELDS_FORMAT_COMPACT;
        let cursor = COMPACT_RECORD_FIELDS_HEADER_BYTES;
        new Int32Array(output.buffer, cursor, scores.length).set(scores);
        cursor += scores.length * 4;
        /* eslint-disable @stylistic/semi */
        const outputOffsets = new Uint32Array(output.buffer, cursor, count)
        cursor += count * 4
        const outputLengths = new Uint16Array(output.buffer, cursor, count)
        if (directReferences !== null) {
            // No await or caller-controlled getter separates these passes.
            // Write final columns directly instead of allocating and copying
            // another six bytes per row of offset/length staging.
            const {uniqueIndexList, offsets: resolvedOffsets, lengths: resolvedLengths} = directReferences
            for (let i = 0; i < count; ++i) {
                const unique = uniqueIndexList[i]
                const offset = resolvedOffsets[unique]
                outputOffsets[i] = offset < 0 ? U32_NULL : offset - contentOffsetBase
                outputLengths[i] = resolvedLengths[unique]
            }
        } else if (offsets !== null && lengths !== null) {
            outputOffsets.set(offsets)
            outputLengths.set(lengths)
        }
        /* eslint-enable @stylistic/semi */
        cursor += count * 2;
        new Uint16Array(output.buffer, cursor, count).set(scoreKeys);
        return output;
    }
}
function selectBaselineWholeRun(chunk) {
 const count=chunk.rowCount,runStart=0,runRowLimit=count,contentOffsets=[],contentLengths=[];
            const useFixedContentSpan = hasFixedContentSpan(chunk, count);
            const references = useFixedContentSpan ? null : getSynchronousResolvedContentReferences(chunk, count);
            let runEnd = runStart;
            const runLimitEnd = Math.min(count, runStart + runRowLimit);
            let minContentOffset = Number.POSITIVE_INFINITY;
            let maxContentOffset = Number.NEGATIVE_INFINITY;
            while (runEnd < runLimitEnd) {
                let uniqueIndex = -1;
                let contentOffset;
                let contentLength;
                if (references !== null) {
                    uniqueIndex = references.uniqueIndexList[runEnd];
                    if (uniqueIndex >= references.offsets.length) {
                        throw new RangeError(`Invalid resolved term content reference index: ${uniqueIndex}`);
                    }
                    contentOffset = references.offsets[uniqueIndex];
                    contentLength = references.lengths[uniqueIndex];
                } else {
                    contentOffset = getArtifactContentOffset(chunk, contentOffsets, runEnd, useFixedContentSpan);
                    contentLength = getArtifactContentLength(chunk, contentLengths, runEnd);
                }
                validateContentOffset(contentOffset);
                validateContentLength(contentLength);
                const nextMinContentOffset = contentOffset >= 0 ? Math.min(minContentOffset, contentOffset) : minContentOffset;
                const nextMaxContentOffset = contentOffset >= 0 ? Math.max(maxContentOffset, contentOffset) : maxContentOffset;
                if (
                    runEnd > runStart &&
                    nextMinContentOffset !== Number.POSITIVE_INFINITY &&
                    (nextMaxContentOffset - nextMinContentOffset) > MAX_CONTENT_OFFSET_DELTA
                ) {
                    break;
                }
                minContentOffset = nextMinContentOffset;
                maxContentOffset = nextMaxContentOffset;
                ++runEnd;
                if (references !== null) {
                    // Consecutive aliases have the same already-validated span
                    // and cannot change this run's minimum or maximum offset.
                    while (runEnd < runLimitEnd && references.uniqueIndexList[runEnd] === uniqueIndex) {
                        ++runEnd;
                    }
                }
            }
            const runContentOffsetBase = minContentOffset === Number.POSITIVE_INFINITY ? 0 : minContentOffset;

return {runEnd, contentOffsetBase:runContentOffsetBase};
}
return {selectBaselineWholeRun,Encoder, factory:createArtifactTermRecordChunkWithResolvedContent, fused:tryEncodeCompactRecordFieldsWithContentBounds, validateContentOffset, validateContentLength, getSynchronousResolvedContentReferences, MAX_CONTENT_OFFSET_DELTA};
